**Step 4: Implement TideMode**

Create `src/modes/tide-mode.ts`:

```ts
import type { GameMode, GameState, WaveParams, WaveOutcome, PhaseTransition } from './game-mode';
import {
  TIDE_BASE_HEIGHT, TIDE_GROWTH_FACTOR, TIDE_EXPONENT,
  MAX_ELEVATION, MIN_ELEVATION,
  ENHANCED_SHOVEL_WAVES_REQUIRED,
} from '../config';

export class TideMode implements GameMode {
  nextWaveParams(state: GameState): WaveParams {
    const waveNumber = state.wavesCompleted + 1;
    return {
      peakHeight: TIDE_BASE_HEIGHT + TIDE_GROWTH_FACTOR * Math.pow(waveNumber, TIDE_EXPONENT),
      waveCount: 1,
    };
  }

  scoopBudget(_state: GameState): number {
    return Infinity;
  }

  elevationBounds(_level: number): { min: number; max: number } {
    return { min: MIN_ELEVATION, max: MAX_ELEVATION };
  }

  resolveWave(_state: GameState, outcome: WaveOutcome): PhaseTransition {
    if (outcome.castleFlooded) {
      return { type: 'gameover' };
    }
    return { type: 'plan' };
  }

  checkCleanWaveReward(state: GameState, isClean: boolean): boolean {
    if (state.hasEnhancedShovel) {
      return false;
    }
    if (!isClean) {
      return false;
    }
    return state.consecutiveCleanWaves + 1 >= ENHANCED_SHOVEL_WAVES_REQUIRED;
  }
}
```

**Step 1: Create TideHud class**

The tide HUD shows:

- **Wave counter**: "Waves: N" text, prominent

Create `src/view/tide-hud.ts`:

```ts
import { Scene, Actor, Color, Text, Font, Rectangle, Vector, Circle, GraphicsGroup, Line } from 'excalibur';
import { CANVAS_WIDTH, TIDE_HIGH_TIDE_WAVE } from '../config';

const HUD_RIGHT_MARGIN = 10;
const HUD_TOP = 4;
const HUD_WIDTH = 260;
const ROW_HEIGHT = 20;
const PADDING_X = 8;
const PADDING_Y = 8;
const Z_BG = 10;
const Z_TEXT = 11;

const CLOCK_RADIUS = 30;
const CLOCK_CENTER_X = 40;
const CLOCK_CENTER_Y = 45;

export class TideHud {
  private bgActor: Actor | null = null;
  private waveCountActor: Actor | null = null;
  private waveCountText: Text | null = null;
  private countdownActor: Actor | null = null;
  private countdownText: Text | null = null;
  private clockHandActor: Actor | null = null;
  private stateActor: Actor | null = null;
  private stateText: Text | null = null;

  private hudX = CANVAS_WIDTH - HUD_WIDTH - HUD_RIGHT_MARGIN;

  activate(scene: Scene): void {
    this.hudX = CANVAS_WIDTH - HUD_WIDTH - HUD_RIGHT_MARGIN;

    const bgHeight = PADDING_Y + ROW_HEIGHT * 4 + PADDING_Y;
    this.bgActor = new Actor({ x: this.hudX, y: HUD_TOP, z: Z_BG, anchor: Vector.Zero });
    this.bgActor.graphics.use(new Rectangle({
      width: HUD_WIDTH,
      height: bgHeight,
      color: Color.fromRGB(0, 0, 0, 0.45),
    }));
    scene.add(this.bgActor);

    const row1Y = HUD_TOP + PADDING_Y + ROW_HEIGHT / 2;
    const row2Y = row1Y + ROW_HEIGHT;
    const row3Y = row2Y + ROW_HEIGHT;

    this.waveCountText = new Text({
      text: 'Waves: 0',
      color: Color.White,
      font: new Font({ size: 16 }),
    });
    this.waveCountActor = new Actor({
      x: this.hudX + PADDING_X,
      y: row1Y,
      z: Z_TEXT,
      anchor: new Vector(0, 0.5),
    });
    this.waveCountActor.graphics.use(this.waveCountText);
    scene.add(this.waveCountActor);

    this.countdownText = new Text({
      text: '',
      color: Color.fromRGB(255, 200, 80),
      font: new Font({ size: 14 }),
    });
    this.countdownActor = new Actor({
      x: this.hudX + PADDING_X,
      y: row2Y,
      z: Z_TEXT,
      anchor: new Vector(0, 0.5),
    });
    this.countdownActor.graphics.use(this.countdownText);
    scene.add(this.countdownActor);

    this.stateText = new Text({
      text: '',
      color: Color.fromRGB(180, 180, 180),
      font: new Font({ size: 12 }),
    });
    this.stateActor = new Actor({
      x: this.hudX + PADDING_X,
      y: row3Y,
      z: Z_TEXT,
      anchor: new Vector(0, 0.5),
    });
    this.stateActor.graphics.use(this.stateText);
    scene.add(this.stateActor);
  }

  updateWaveCount(count: number): void {
    if (this.waveCountText && this.waveCountActor) {
      this.waveCountText.text = `Waves: ${count}`;
      this.waveCountActor.graphics.use(this.waveCountText);
    }
  }

  updateCountdown(secondsLeft: number): void {
    if (this.countdownText && this.countdownActor) {
      this.countdownText.text = secondsLeft > 0 ? `Next wave: ${secondsLeft}s` : 'WAVE!';
      this.countdownActor.graphics.use(this.countdownText);
    }
  }

  updateState(text: string): void {
    if (this.stateText && this.stateActor) {
      this.stateText.text = text;
      this.stateActor.graphics.use(this.stateText);
    }
  }

  deactivate(scene: Scene): void {
    for (const actor of [this.bgActor, this.waveCountActor, this.countdownActor, this.stateActor, this.clockHandActor]) {
      if (actor) {
        scene.remove(actor);
      }
    }
    this.bgActor = null;
    this.waveCountActor = null;
    this.waveCountText = null;
    this.countdownActor = null;
    this.countdownText = null;
    this.stateActor = null;
    this.stateText = null;
    this.clockHandActor = null;
  }
}
```

### Task 4: TideSession scene (core game loop)

**Files:**
- Create: `src/tide-session.ts`

This is the largest task. TideSession is a new Scene that runs the tide mode game loop. It reuses GridModel, GridView, WaveRenderer, and the modified PlanningPhase.

**Step 1: Create TideSession**

Create `src/tide-session.ts`. The structure mirrors `GameSession` but with key differences:
- No level concept (level is always 1 in GameState, wavesCompleted tracks progress)
- PlanningPhase is activated once and stays active (unlimited scoops, no Send Wave button)
- Waves fire on a repeating timer
- Uses TideHud instead of Hud
- Game over shows waves survived instead of level reached

```ts
import { Engine, Scene, Actor, Keys, vec } from 'excalibur';
import { GridView } from './view/grid-view';
import { GridModel } from './model/grid-model';
import { PlanningPhase } from './view/planning-phase';
import { WaveRenderer } from './view/wave-renderer';
import { TideHud } from './view/tide-hud';
import {
  showWaveBanner, showTextBanner, showGameOver,
  showElevationLabels, hideElevationLabels,
} from './view/screen-overlays';
import { simulateWave, generateWaveCurve } from './model/wave-simulation';
import {
  GRID_HEIGHT, TILE_SIZE, TERRAIN_SLOPE,
  CASTLE_ROW, CASTLE_COL, GRID_WIDTH,
  GRID_TOP, GRID_LEFT,
  WAVE_VALLEY_FRACTION, WAVE_PEAK_WEIGHTS,
  TIDE_WAVE_INTERVAL_MS,
} from './config';
import type { GameState } from './modes/game-mode';
import { TideMode } from './modes/tide-mode';
import { Tile } from './view/tile';
import { tiledMap } from './resources';

export class TideSession extends Scene {
  private model!: GridModel;
  private grid!: GridView;
  private waveRenderer!: WaveRenderer;
  private hud!: TideHud;
  private planningPhase!: PlanningPhase;
  private elevationLabelActors: Actor[] = [];
  private gameMode = new TideMode();
  private state: GameState = {
    level: 1,
    wavesCompleted: 0,
    consecutiveCleanWaves: 0,
    hasEnhancedShovel: false,
  };
  private waveTimer: ReturnType<typeof setInterval> | null = null;
  private countdownTimer: ReturnType<typeof setInterval> | null = null;
  private waveInProgress = false;
  private secondsUntilWave = 0;

  override onInitialize(_engine: Engine): void {
    const TILED_TILE_SIZE = 16;
    const tileScale = TILE_SIZE / TILED_TILE_SIZE;
    const TILEMAP_OCEAN_ROWS = 6;
    const mapX = GRID_LEFT;
    const mapY = GRID_TOP - TILEMAP_OCEAN_ROWS * TILE_SIZE;
    tiledMap.addToScene(this);
    for (const layer of tiledMap.getTileLayers()) {
      const tm = layer.tilemap;
      tm.pos = vec(mapX, mapY);
      tm.scale = vec(tileScale, tileScale);
      tm.z = -1;
    }

    this.model = new GridModel({
      width: GRID_WIDTH,
      height: GRID_HEIGHT,
      castleCol: CASTLE_COL,
      castleRow: CASTLE_ROW,
    });
    this.grid = new GridView(this.model, this);
    this.waveRenderer = new WaveRenderer(this.grid, this);
    this.hud = new TideHud();
    this.hud.activate(this);

    this.startPlanning();
    this.startWaveTimer();

    _engine.input.keyboard.on('hold', (evt) => {
      if (evt.key === Keys.L && this.elevationLabelActors.length === 0) {
        this.elevationLabelActors = showElevationLabels(this, this.grid);
      }
    });
    _engine.input.keyboard.on('release', (evt) => {
      if (evt.key === Keys.L) {
        hideElevationLabels(this, this.elevationLabelActors);
        this.elevationLabelActors = [];
      }
    });
    _engine.input.keyboard.on('press', (evt) => {
      if (evt.key === Keys.D) {
        const text = this.model.serialize();
        void navigator.clipboard.writeText(text);
      }
    });
  }

  private startPlanning(): void {
    const waveParams = this.gameMode.nextWaveParams(this.state);
    const naturalReach = Math.min(
      Math.round(waveParams.peakHeight / TERRAIN_SLOPE),
      GRID_HEIGHT,
    );
    this.planningPhase = new PlanningPhase(
      this.grid,
      this.hud,
      Infinity,
      naturalReach,
      waveParams.peakHeight,
      1,
      this.state.hasEnhancedShovel,
      () => {},
    );
    this.planningPhase.activate(this);
  }

  private startWaveTimer(): void {
    this.secondsUntilWave = Math.round(TIDE_WAVE_INTERVAL_MS / 1000);
    this.hud.updateCountdown(this.secondsUntilWave);

    this.countdownTimer = setInterval(() => {
      this.secondsUntilWave--;
      this.hud.updateCountdown(Math.max(0, this.secondsUntilWave));
    }, 1000);

    this.waveTimer = setInterval(() => {
      if (!this.waveInProgress) {
        void this.runWave();
      }
    }, TIDE_WAVE_INTERVAL_MS);
  }

  private stopTimers(): void {
    if (this.waveTimer) {
      clearInterval(this.waveTimer);
      this.waveTimer = null;
    }
    if (this.countdownTimer) {
      clearInterval(this.countdownTimer);
      this.countdownTimer = null;
    }
  }

  private async runWave(): Promise<void> {
    this.waveInProgress = true;
    this.secondsUntilWave = Math.round(TIDE_WAVE_INTERVAL_MS / 1000);

    const waveParams = this.gameMode.nextWaveParams(this.state);
    const waveHeight = waveParams.peakHeight;

    const banner = showWaveBanner(this, this.state.wavesCompleted + 1, 0);
    await this.delay(500);
    this.remove(banner);

    const peakPhase = (Math.random() - 0.5) * 0.4;
    const totalWeight = WAVE_PEAK_WEIGHTS.reduce((a, b) => a + b, 0);
    let r = Math.random() * totalWeight;
    let numPeaks = 1;
    for (let i = 0; i < WAVE_PEAK_WEIGHTS.length; i++) {
      r -= WAVE_PEAK_WEIGHTS[i];
      if (r <= 0) {
        numPeaks = i + 1;
        break;
      }
    }
    const columnHeights = generateWaveCurve(
      GRID_WIDTH, waveHeight, WAVE_VALLEY_FRACTION, peakPhase, numPeaks,
    );

    const elevations = this.grid.model.getElevations();
    const puddleDepths: number[][] = elevations.map((row, rowIdx) =>
      row.map((_, colIdx) => this.grid.model.getPuddleDepth(colIdx, rowIdx)),
    );

    const result = simulateWave({
      elevations,
      puddleDepths,
      columnHeights,
      castleCol: CASTLE_COL,
      castleRow: CASTLE_ROW,
      maxRows: GRID_HEIGHT,
      terrainSlope: TERRAIN_SLOPE,
      poolMap: this.grid.model.getPoolMap(),
    });

    await this.waveRenderer.playWave(result);

    const erodedTiles = this.grid.applyErosion(
      result.advanceHeightMap, result.recedeHeightMap,
    );
    if (erodedTiles.length > 0) {
      await this.waveRenderer.flashErodedTiles(erodedTiles);
    }

    const puddleDeltas: { col: number; row: number; depth: number }[] = [];
    for (let pr = 0; pr < result.puddleDelta.length; pr++) {
      for (let c = 0; c < result.puddleDelta[pr].length; c++) {
        if (result.puddleDelta[pr][c] > 0) {
          puddleDeltas.push({ col: c, row: pr, depth: result.puddleDelta[pr][c] });
        }
      }
    }
    this.grid.applyPuddleDeltas(puddleDeltas);
    this.grid.applySandRedistribution(result.wallErosionEvents);
    await this.waveRenderer.flashSandRedistribution(result.wallErosionEvents);

    this.state.wavesCompleted++;
    this.hud.updateWaveCount(this.state.wavesCompleted);

    const transition = this.gameMode.resolveWave(this.state, {
      castleFlooded: result.castleFlooded,
      allWavesComplete: true,
    });

    if (transition.type === 'gameover') {
      this.stopTimers();
      this.planningPhase.deactivate(this);
      this.waveRenderer.cleanup();
      showGameOver(this, this.state.wavesCompleted, {
        onRestart: () => this.resetGame(),
      });
      return;
    }

    this.waveRenderer.cleanup();
    this.waveInProgress = false;
  }

  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }

  private resetGame(): void {
    this.state = {
      level: 1,
      wavesCompleted: 0,
      consecutiveCleanWaves: 0,
      hasEnhancedShovel: false,
    };
    this.hud.updateWaveCount(0);
    this.waveRenderer.cleanup();
    const tilesToRemove = this.entities.filter((e) => e instanceof Tile) as Tile[];
    for (const tile of tilesToRemove) {
      this.remove(tile);
    }
    this.model = new GridModel({
      width: GRID_WIDTH,
      height: GRID_HEIGHT,
      castleCol: CASTLE_COL,
      castleRow: CASTLE_ROW,
    });
    this.grid = new GridView(this.model, this);
    this.waveRenderer = new WaveRenderer(this.grid, this);
    this.startPlanning();
    this.waveInProgress = false;
    this.startWaveTimer();
  }

  override onDeactivate(): void {
    this.stopTimers();
  }
}
```
